cmake_minimum_required(VERSION 3.20)
project(ClaudeDesktopChineseInjector LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
add_library(claude_zh_hook SHARED native/hook.cpp)
target_link_libraries(claude_zh_hook PRIVATE minhook)
target_compile_definitions(claude_zh_hook PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(claude_zh_hook PRIVATE /utf-8)
set_target_properties(claude_zh_hook PROPERTIES OUTPUT_NAME "claude-zh-hook")
add_executable(claude_zh_launcher WIN32 native/launcher.cpp)
target_compile_definitions(claude_zh_launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(claude_zh_launcher PRIVATE /utf-8)
set_target_properties(claude_zh_launcher PROPERTIES OUTPUT_NAME "ClaudeChineseLauncher")
